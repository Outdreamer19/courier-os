const params = new URLSearchParams(typeof window !== 'undefined' ? window.location.search : '');
const format = (params.get('format') ?? 'story') as 'story' | 'feed' | 'og';
const fixedScene = params.get('scene');
const reel = params.get('reel') === '1';

const size = computed(() =>
    format === 'feed' ? { w: 1080, h: 1350 } : format === 'og' ? { w: 1200, h: 630 } : { w: 1080, h: 1920 },

const durations: Record<number, number> = { 1: 4200, 2: 6800, 3: 4200, 4: 4200, 5: 5000 };

    if (!reel) return;
    const next = () => {
        if (scene.value >= 5) return;
        setTimeout(() => {
            scene.value += 1;
            next();
        }, durations[scene.value]);
    };
    next();
});
</script>
